\node[card, draw=codegreen!58, text width=1.30cm] (positives) at (2.75,2.20)

  {{\tiny\bfseries Relational}\\[-1pt]
   {\scriptsize \((x,y){=}(2,3)\;\mapsto\;(\textcolor{wrongred}{3},3)\)}};

\node[card, draw=codepurple!48, text width=2.75cm]
  (post) at (6.85,2.20)
  {{\tiny\bfseries Post-exit}\\[-1pt]
   {\scriptsize \(s_{\mathrm{exit}}\!\rightarrow\!
   \textcolor{wrongred}{s_1^-\!\rightarrow\!s_2^-\cdots}\)}};

\node[card, draw=codepurple!48, text width=2.75cm]
  (range) at (6.85,1.40)
  {{\tiny\bfseries Range}\\[-1pt]

\node[card, draw=codeblue!54, text width=1.30cm]

\node[card, draw=codegreen!62, text width=0.85cm]

\caption{\textbf{Construction of target-hidden negative traces.}
Concrete executions provide observed reachable states.  Three transformations
probe relational structure, behavior past a genuine exit, and sampled bounds.
Conservative cleaning and per-family budgets produce \(\mathcal N\), which is
used only for RL reward computation and never exposes \(Q\).}

\node[card, draw=paperink!34, text width=1.50cm] (execute) at (1.00,2.45)
  {{\tiny\bfseries Execute}\\[-1pt]
   {\tiny input sweep \(U\)}\\[-1pt]
   {\tiny target \(Q\) hidden}};

\node[card, draw=codegreen!58, fill=codegreen!8, text width=1.95cm]
  (positives) at (3.45,2.45)
  {{\tiny\bfseries Observed trace}\\[-1pt]
   {\scriptsize\textcolor{codegreen!82}{\(\bullet\!\rightarrow\!\bullet\!
   \!\rightarrow\!\bullet\)}}\\[-1pt]
   {\tiny positives \(\mathcal P\)}};

\draw[flow] (execute) -- (positives);

%% -----------------------------------------------------------------
%% Three candidate families (panel)
%% -----------------------------------------------------------------
\draw[rounded corners=5pt, fill=codepurple!2, draw=codepurple!34,
  line width=0.8pt] (5.10,0.30) rectangle (10.10,4.60);
\node[tag, fill=codepurple!84, anchor=west] at (5.32,4.30)
  {THREE CANDIDATE FAMILIES};

\node[card, draw=codepurple!48, fill=codepurple!4, text width=3.70cm]
  (rel) at (7.60,3.63)
  {{\tiny\bfseries Relational perturbation}\\[-1pt]
   {\scriptsize \((x,y)=(2,3)\;\mapsto\;(3,3)\)}\\[-1pt]
   {\tiny small local edit breaks \(x{+}y{=}5\)}};

\node[card, draw=codepurple!48, fill=codepurple!4, text width=3.70cm]
  (post) at (7.60,2.45)
  {{\tiny\bfseries Post-exit continuation}\\[-1pt]
   {\scriptsize \(s_{\mathrm{exit}}\!\rightarrow\!s_1^-\!
   \rightarrow\!s_2^-\cdots\)}\\[-1pt]
   {\tiny continue past a genuine exit}};

\node[card, draw=codepurple!48, fill=codepurple!4, text width=3.70cm]
  (range) at (7.60,1.27)
  {{\tiny\bfseries Range perturbation}\\[-1pt]
   {\scriptsize observed \(x\in[1,3]\;\mapsto\;x=8\)}\\[-1pt]
   {\tiny move \(x\) just outside its range}};

\node[note] at (7.60,0.02)
  {relation/range: one state \(=\) one trace\\
   post-exit: the whole continuation \(=\) one trace};

\coordinate (fork) at (4.92,2.45);
\draw[branch] (positives.east) -- (fork);
\draw[candidateflow] (fork) |- (rel.west);
\draw[candidateflow] (fork) -- (post.west);
\draw[candidateflow] (fork) |- (range.west);

%% -----------------------------------------------------------------
%% Merge -> conservative cleaning -> per-family budgets -> N
%% -----------------------------------------------------------------
\coordinate (merge) at (10.30,2.45);
\draw[branch] (rel.east) -| (merge);
\draw[branch] (post.east) -- (merge);
\draw[branch] (range.east) -| (merge);

\node[card, draw=codered!48, fill=codered!3, text width=1.85cm]
  (clean) at (11.95,2.45)
  {{\tiny\bfseries Conservative clean}\\[-1pt]
   {\tiny remove observed states,}\\[-2pt]
   {\tiny fresh entries, duplicates,}\\[-2pt]
   {\tiny unsupported candidates}};
\draw[candidateflow] (merge) -- (clean);

\node[card, draw=codeblue!54, fill=algobg!68, text width=2.05cm]
  (budget) at (15.00,2.45)
  {{\tiny\bfseries Family budgets}\\[-1pt]
   {\tiny stable round-robin}\\[-2pt]
   {\tiny rel \(\cdot\) post \(\cdot\) range}};
\draw[flow] (clean) -- (budget);

\node[card, draw=codegreen!62, fill=codegreen!9, text width=1.60cm]
  (negatives) at (17.90,2.45)
  {{\tiny\bfseries Negative traces}\\[-1pt]

   {\tiny singleton states}\\[-2pt]
   {\tiny + continuation traces}};
\draw[flow] (budget) -- (negatives);

\node[note, text=codered!78] at (14.30,0.95)
  {reward signal only; not a proof of unreachability};

\caption{\textbf{Construction of target-hidden negative traces.}
Concrete executions provide observed reachable states.  Three transformations
probe relational structure, behavior past a genuine exit, and sampled bounds.
Conservative cleaning and per-family budgets produce \(\mathcal N\), which is
used only for RL reward computation and never exposes \(Q\).}
